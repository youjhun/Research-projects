def create_lagged_features(X_raw, Y_vx, Y_vy, num_lags=NUM_LAGS):
    N, D = X_raw.shape
    shape   = (N - num_lags, num_lags, D)
    strides = (X_raw.strides[0], X_raw.strides[0], X_raw.strides[1])

    vy_lag  = Y_vy[num_lags:].astype(np.float32)
    return X_lag, vx_lag, vy_lag

def create_sequential_features(X_raw, num_lags=NUM_LAGS):

def apply_dynamic_weighting(X_feat, burst_power=BURST_POWER):

def apply_SER(spike_matrix, ser, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    corrupted = spike_matrix.copy().astype(float)
    half_ser  = ser / 2.0
    spike_idx = np.argwhere(corrupted > 0)
    if len(spike_idx) > 0 and half_ser > 0:
        n_miss = int(round(len(spike_idx) * half_ser))
        if n_miss > 0:
            chosen = rng.choice(len(spike_idx), size=n_miss, replace=False)
            corrupted[spike_idx[chosen, 0], spike_idx[chosen, 1]] = 0
    zero_idx = np.argwhere(corrupted == 0)
    if len(zero_idx) > 0 and half_ser > 0:
        n_false = int(round(int(np.sum(spike_matrix > 0)) * half_ser))
        n_false = min(n_false, len(zero_idx))
        if n_false > 0:
            chosen = rng.choice(len(zero_idx), size=n_false, replace=False)
            corrupted[zero_idx[chosen, 0], zero_idx[chosen, 1]] = 1
    return corrupted

def evaluate_decoding(y_true, y_pred):
    r, _       = pearsonr(y_true, y_pred)
    vaf        = float(1 - np.var(y_true - y_pred) / np.var(y_true))
    rmse       = float(np.sqrt(np.mean((y_true - y_pred) ** 2)))
    scale_bias = float(np.std(y_pred) / np.std(y_true))
    return {'r': r, 'vaf': vaf, 'rmse': rmse, 'scale_bias': scale_bias}

def create_peak_aware_features(X_raw, Y_vx, Y_vy, num_lags=NUM_LAGS, pca_components=PCA_COMPONENTS):
    X_lag, vx_lag, vy_lag = create_lagged_features(X_raw, Y_vx, Y_vy, num_lags)
    N, D = X_lag.shape
    n_neurons = X_raw.shape[1]
    X_3d = X_lag.reshape(N, num_lags, n_neurons)

    pop_activity = X_3d.sum(axis=2)               
    burst_score  = pop_activity.max(axis=1, keepdims=True)    
    burst_mean   = pop_activity.mean(axis=1, keepdims=True)   
    max_spike    = X_3d.max(axis=(1,2), keepdims=False).reshape(N, 1)  
    active_ratio = (X_3d > 0).mean(axis=(1,2), keepdims=False).reshape(N, 1)

    recent_act   = X_3d[:, -5:, :].sum(axis=(1,2), keepdims=False).reshape(N,1)
    old_act      = X_3d[:, :-5, :].sum(axis=(1,2), keepdims=False).reshape(N,1) + 1e-6
    onset_ratio  = recent_act / old_act            

    peak_features = np.hstack([
        burst_score, burst_mean, max_spike,
        active_ratio, onset_ratio
    ]).astype(np.float32)                              

    return X_lag, peak_features, vx_lag, vy_lag

# ─── Ridge ────────────────────────────────────────────────────
def fit_ridge_gpu(X_tr, Y_tr, alpha=RIDGE_ALPHA):
    N, D = X_tr.shape
    A = torch.matmul(X_tr.T, X_tr) + alpha * torch.eye(D, device=DEVICE)
    B = torch.matmul(X_tr.T, Y_tr)          
    return torch.linalg.solve(A, B)          

# ─── PeakAwareMLP & PCA ───────────────────────────────────────
def fit_pca_gpu(X_tr, n_components=PCA_COMPONENTS):
    X_mean = X_tr.mean(dim=0)
    _, _, V = torch.pca_lowrank(X_tr - X_mean, q=n_components, center=False)
    return X_mean, V

class PeakAwareLoss(nn.Module):
    def __init__(self, peak_threshold=8.0, alpha=3.0, delta=1.0):
        super().__init__()
        self.peak_thr = peak_threshold
        self.alpha    = alpha
        self.huber    = nn.HuberLoss(reduction='none', delta=delta)

    def forward(self, pred, target):
        base_loss = self.huber(pred, target)          

        peak_mask = (target.abs() > self.peak_thr).float()
        weights   = 1.0 + (self.alpha - 1.0) * peak_mask  

        grad = torch.diff(target, dim=0).abs()
        grad = torch.cat([grad[:1], grad], dim=0)    
        grad_w = 1.0 + grad / (grad.mean() + 1e-8)  

        return (base_loss * weights * grad_w).mean()

class PeakAwareMLP(nn.Module):
    def __init__(self, raw_dim, peak_dim=5, hidden_dims=MLP_HIDDEN):
        super().__init__()
        
        # 메인 피처 스트림 (512 ➔ 256 차원으로 비선형 맵핑)
        self.pca_stream = nn.Sequential(
            nn.Linear(raw_dim, hidden_dims[0]), nn.ReLU(),
            nn.Linear(hidden_dims[0], hidden_dims[1]), nn.ReLU(),
        )
        
        # Peak 피처 스트림 (기존 유지)
        self.peak_stream = nn.Sequential(
            nn.Linear(peak_dim, 32), nn.ReLU(),
            nn.Linear(32, 32), nn.ReLU(),
        )
        
        # 융합 레이어 ( (256 + 32) ➔ 128 차원으로 압축 후 최종 2차원 속도 예측 )
        self.fusion = nn.Sequential(
            nn.Linear(hidden_dims[1] + 32, hidden_dims[2]), nn.ReLU(),
            nn.Linear(hidden_dims[2], 2)
        )

    def forward(self, x_pca, x_peak):
        h1 = self.pca_stream(x_pca)
        h2 = self.peak_stream(x_peak)
        return self.fusion(torch.cat([h1, h2], dim=1))     

def train_mlp_gpu(X_pca, X_peak, Y_tr, epochs=MLP_EPOCHS):
    # [수정] hidden_dims 인자를 명시적으로 주입하여 스케일 업 반영
    model  = PeakAwareMLP(raw_dim=X_pca.shape[1], peak_dim=X_peak.shape[1], hidden_dims=MLP_HIDDEN).to(DEVICE)
    opt    = optim.Adam(model.parameters(), lr=LEARNING_RATE, weight_decay=MLP_ALPHA)
    crit   = PeakAwareLoss(peak_threshold=8.0, alpha=3.0)   
    
    loader = DataLoader(
        TensorDataset(X_pca, X_peak, Y_tr),
        batch_size=BATCH_SIZE,
        shuffle=True,
        pin_memory=False,          
        num_workers=0,
        persistent_workers=False
    )
    model.train()
    for _ in range(epochs):
        for bx_pca, bx_peak, by in loader:
            opt.zero_grad(set_to_none=True)     
            loss = crit(model(bx_pca, bx_peak), by)
            loss.backward()
            opt.step()
    model.eval()
    return model

# ─── LSTM ────────────────────────────────────────────────────
class LSTMDecoder(nn.Module):
    def __init__(self, input_dim, output_dim=2):
        super().__init__()
        self.lstm = nn.LSTM(
            input_size=input_dim,
            hidden_size=LSTM_HIDDEN,
            num_layers=LSTM_LAYERS,
            batch_first=True,
            dropout=LSTM_DROPOUT if LSTM_LAYERS > 1 else 0.0
        )
        self.fc = nn.Linear(LSTM_HIDDEN, output_dim)

    def forward(self, x):
        out, _ = self.lstm(x)
        return self.fc(out[:, -1, :])   

def train_lstm_gpu(X_tr, Y_tr, epochs=LSTM_EPOCHS):
    model  = LSTMDecoder(input_dim=X_tr.shape[2]).to(DEVICE)
    opt    = optim.Adam(model.parameters(), lr=LEARNING_RATE, weight_decay=MLP_ALPHA)
    crit   = nn.MSELoss()
    loader = DataLoader(
        TensorDataset(X_tr, Y_tr),
        batch_size=BATCH_SIZE,
        shuffle=True,
        num_workers=0
    )
    model.train()
    for _ in range(epochs):
        for bx, by in loader:
            opt.zero_grad(set_to_none=True)
            loss = crit(model(bx), by)
            loss.backward()
            opt.step()
    model.eval()
    return model

# ─── Kalman ────────────────────────────────────────────────────
class KalmanDecoder:
    def __init__(self, n_riccati_iter=KALMAN_ITER):
        self.n_riccati_iter = n_riccati_iter
        
    def fit(self, X_train, Vx_train, Vy_train):
        Z = np.column_stack([Vx_train, Vy_train])
        self.A = np.linalg.lstsq(Z[:-1], Z[1:], rcond=None)[0].T

        W_xz = np.linalg.lstsq(X_train, Z, rcond=None)[0]  
        alpha_H = 1.0
        HtH = Z.T @ Z + alpha_H * np.eye(2)
        self.H = (np.linalg.solve(HtH, Z.T @ X_train)).T  

        res_Q = Z[1:] - (self.A @ Z[:-1].T).T
        self.Q = (res_Q.T @ res_Q) / len(res_Q) + 1e-6 * np.eye(2)

        res_R = X_train - (self.H @ Z.T).T
        raw_var = np.var(res_R, axis=0)
        spike_var_floor = np.var(X_train, axis=0) * 0.1 + 1e-4
        self.R_diag = np.maximum(raw_var, spike_var_floor)

        HtRinv = self.H.T / self.R_diag
        HtRinvH = HtRinv @ self.H

        P = np.diag([np.var(Vx_train), np.var(Vy_train)])

        for _ in range(self.n_riccati_iter):
            K = np.linalg.inv(
                np.linalg.inv(P) + HtRinvH + 1e-8 * np.eye(2)
            ) @ HtRinv
            P_upd = (np.eye(2) - K @ self.H) @ P
            P = self.A @ P_upd @ self.A.T + self.Q

        self.K_steady = K
        return self
    
    def predict(self, X_test):
        preds = []
        z = np.zeros(2)
        for t in range(len(X_test)):
            z_prior = self.A @ z
            z = z_prior + self.K_steady @ (X_test[t] - self.H @ z_prior)
            preds.append(z)
        return np.array(preds)

def run_gpu_decoding(model_type, X_feat, Vx_tgt, Vy_tgt, X_peak=None,
                     n_splits=N_FOLDS, sigma=GAUSS_SIGMA):
    kf = KFold(n_splits=n_splits, shuffle=False)
    Y_combined = np.stack([Vx_tgt, Vy_tgt], axis=1)

    # 🚀 [수정] PCA 관련 글로벌 변수(V_global, pmean_global) 제거
    xmean_global = xstd_global = None
    pfeat_mean_global = pfeat_std_global = None
    
    if model_type == 'MLP':
        # 메인 피처 글로벌 정규화용 통계치만 계산 (PCA는 제거)
        X_all_t = torch.tensor(X_feat, dtype=torch.float32, device=DEVICE)
        xmean_global = X_all_t.mean(dim=0, keepdim=True)
        xstd_global  = X_all_t.std(dim=0, keepdim=True) + 1e-8
        
        # [추가] Peak 피처 글로벌 정규화 (기존 유지)
        if X_peak is not None:
            P_all_t = torch.tensor(X_peak, dtype=torch.float32, device=DEVICE)
            pfeat_mean_global = P_all_t.mean(dim=0, keepdim=True)
            pfeat_std_global  = P_all_t.std(dim=0, keepdim=True) + 1e-8
            del P_all_t

        del X_all_t
        torch.cuda.empty_cache()

    r_x_list, r_y_list, metrics_list = [], [], []
    last_preds = None
    t_start = time.time()

    for train_idx, test_idx in kf.split(X_feat):
        vxte_np = Vx_tgt[test_idx]
        vyte_np = Vy_tgt[test_idx]

        if model_type == 'Kalman':
            kalman = KalmanDecoder()
            kalman.fit(X_feat[train_idx], Vx_tgt[train_idx], Vy_tgt[train_idx])
            p2d    = kalman.predict(X_feat[test_idx])
            vxp_np = gaussian_filter1d(p2d[:, 0], sigma=sigma)
            vyp_np = gaussian_filter1d(p2d[:, 1], sigma=sigma)

        else:
            Xtr = torch.tensor(X_feat[train_idx], dtype=torch.float32, device=DEVICE)
            Xte = torch.tensor(X_feat[test_idx],  dtype=torch.float32, device=DEVICE)
            Ytr = torch.tensor(Y_combined[train_idx], dtype=torch.float32, device=DEVICE)

            if model_type == 'Ridge':
                W   = fit_ridge_gpu(Xtr, Ytr)
                out = torch.matmul(Xte, W)

            elif model_type == 'MLP':
                # 1. 메인 피처 정규화
                Xtr_n = (Xtr - xmean_global) / xstd_global
                Xte_n = (Xte - xmean_global) / xstd_global
                
                # 🚀 [수정] torch.matmul을 통한 PCA 투영(Xtr_p, Xte_p) 코드 완벽 삭제!
                
                # 2. Peak Feature 텐서화 및 정규화 (기존 유지)
                Xtr_peak_raw = torch.tensor(X_peak[train_idx], dtype=torch.float32, device=DEVICE)
                Xte_peak_raw = torch.tensor(X_peak[test_idx],  dtype=torch.float32, device=DEVICE)
                
                Xtr_peak_n = (Xtr_peak_raw - pfeat_mean_global) / pfeat_std_global
                Xte_peak_n = (Xte_peak_raw - pfeat_mean_global) / pfeat_std_global
                
                # 🚀 [수정] Xtr_p, Xte_p 대신 원본 정보가 다 살아있는 정규화 피처 Xtr_n, Xte_n을 직접 대입!
                mlp = train_mlp_gpu(Xtr_n, Xtr_peak_n, Ytr)
                with torch.no_grad():
                    out = mlp(Xte_n, Xte_peak_n)

            elif model_type == 'LSTM':
                xmean = Xtr.mean(dim=(0,1), keepdim=True)
                xstd  = Xtr.std(dim=(0,1),  keepdim=True) + 1e-8
                lstm  = train_lstm_gpu((Xtr-xmean)/xstd, Ytr)
                with torch.no_grad():
                    out = lstm((Xte-xmean)/xstd)

            vxp_np = gaussian_filter1d(out[:,0].cpu().numpy(), sigma=sigma)
            vyp_np = gaussian_filter1d(out[:,1].cpu().numpy(), sigma=sigma)

        rx, _ = pearsonr(vxte_np, vxp_np)
        ry, _ = pearsonr(vyte_np, vyp_np)
        r_x_list.append(rx)
        r_y_list.append(ry)
        metrics_list.append({
            'x': evaluate_decoding(vxte_np, vxp_np),
            'y': evaluate_decoding(vyte_np, vyp_np)
        })
        last_preds = (vxte_np, vxp_np, vyte_np, vyp_np)

    return r_x_list, r_y_list, last_preds, (time.time()-t_start), metrics_list

## 1. 공통 유틸리티 함수 (Feature Engineering & Error Simulation)

### `def create_lagged_features(X_raw, Y_vx, Y_vy, num_lags=NUM_LAGS)`

- **역할**: 과거 신경 신호 이력(역사)을 한 시점에 모아 다차원 피처로 평탄화(Flatten)합니다.
- **공학적 기법 (`Stride Tricks`)**: 메모리를 새로 할당해 복사하는 대신, 배열의 물리적 포인터 주소 점프 폭(`strides`)만 O(1) 연산으로 변경하여 Colab 환경에서 메모리 고갈(OOM)을 원천 차단합니다.
- **I/O**:
    - **Input**: `X_raw` [N, Neurons], `Y_vx` [N], `Y_vy` [N]
    - **Output**: `X_lag` [N - 20, 20 $\times$ Neurons], `vx_lag` [N - 20], `vy_lag` [N - 20]

### `def create_sequential_features(X_raw, num_lags=NUM_LAGS)`

- **역할**: 순차 모델인 `LSTM`에 주입하기 위해 시간 축이 살아있는 3차원 텐서 형태의 시계열 윈도우를 생성합니다.
- **I/O**:
    - **Input**: `X_raw` [N, Neurons]
    - **Output**: `X_seq` [N - 20, 20, Neurons] (윈도우 크기 20 유지)

### `def get_kalman_obs(X_raw, num_lags=NUM_LAGS)`

- **역할**: 칼만 필터(`KalmanDecoder`) 모델을 위한 스파이크 관측 행렬을 만듭니다. 칼만 필터는 상태공간 모델의 마르코프 성질(Markov Property)에 의해 과거 이력 없이 현재 시점(`t`)의 1차원 스파이크 정보만 필요하므로, 다른 모델과의 시간 축 정렬을 위해 래깅 구간(`num_lags`)만큼 앞쪽만 잘라내어 반환합니다.
- **I/O**:
    - **Input**: `X_raw` [N, Neurons]
    - **Output**: `X_obs` [N - 20, Neurons]

### `def apply_dynamic_weighting(X_feat, burst_power=BURST_POWER)`

- **생물학적 배경**: 동물이 방향을 급격히 틀거나 속도를 낼 때, 뉴런은 스파이크를 기관총처럼 연사하는 '버스트(Burst) 발화'를 일으킵니다. 실제 근육의 가속도는 이 스파이크 빈도에 비선형적(지수적)으로 반응합니다.
- **역할**: 선형 모델이나 일반 딥러닝이 최고 속도 구간에서 예측치가 찌그러지는 현상(Undershoot)을 막기 위해, 스파이크가 발생한 곳(`> 0`)에 지수 연산(`BURST_POWER = 1.3`)을 가해 발화 강도를 비선형적으로 증폭시킵니다.
- **I/O**:
    - **Input / Output**: 동일한 형태의 변형된 피처 Matrix

### `def apply_SER(spike_matrix, ser, rng=None)`

- **시스템적 배경**: 무선 신경 임플란트 칩에서 수신기(Back-end)로 데이터를 보낼 때, 무선 통신 환경 노이즈로 인해 데이터 파편이 손실되거나 튀는 스파이크 에러율(SER: Spike Error Rate)이 발생합니다.
- **역할**: 스파이크가 있었던 자리를 지우는 누락 오차(Miss, `0`으로 변환)와, 스파이크가 없던 정적 자리에 가짜 스파이크가 노이즈로 박히는 오검출 오차(False Alarm, `1`로 변환)를 통계적으로 주입하여 통신 환경을 시뮬레이션합니다. (본 기저 성능 비교 코드에서는 선언만 되어 있고 PHASE 2 기저선 검증을 위해 대기 중입니다.)

### `def evaluate_decoding(y_true, y_pred)`

- **역할**: 복원된 운동 신경 궤적의 퀄리티를 평가하기 위해 고전적인 상관계수(`r`), 분산 설명도(`vaf`), 오차 절대 폭(`rmse`), 그리고 예측 궤적이 실제 크기 대비 얼마나 위축되었는지 보여주는 진폭 왜곡 지표(`scale_bias`) 등 BCI 저널 표준 평가지표 4종을 계산합니다.
- **I/O**:
    - **Input**: `y_true` [N], `y_pred` [N]
    - **Output**: 평가지표 딕셔너리 (`r`, `vaf`, `rmse`, `scale_bias`)

### `def create_peak_aware_features(X_raw, Y_vx, Y_vy, num_lags=NUM_LAGS)`

- **역할**: 일반적인 스파이크 행렬 외에, 전체 신경망의 거시적 변동성을 포착해 `MLP` 모델에게 이정표로 제공할 5가지 핵심 거시 피처를 별도로 정밀 추출합니다.
- **추출되는 5가지 피처**:
    1. `burst_score`: 현재 윈도우(200ms) 내에서 전체 뉴런 발화 합산량의 최댓값
    2. `burst_mean`: 윈도우 내 평균 앙상블 발화량
    3. `max_spike`: 단일 뉴런이 기록한 최대 발화량
    4. `active_ratio`: 윈도우 내에서 활성화된 뉴런의 공간적 비율 (Sparsity 역산)
    5. `onset_ratio`: 최근 50ms의 발화량을 이전 150ms 발화량으로 나눈 값 (속도 변화의 '시작점/온셋'을 감지하는 누적 가속 지표)
- **I/O**:
    - **Input**: 원본 데이터스트림
    - **Output**: `X_lag` [N-20, 20 $\times$ Neurons] 및 고차원 거시 피처 행렬 `peak_features` [N-20, 5]

## 2. 모델 구현 계층 (Model Architectures)

### `def fit_ridge_gpu(X_tr, Y_tr, alpha=RIDGE_ALPHA)`

- **역할**: PyTorch GPU 가속 선형 대수 연산(`torch.linalg.solve`)을 이용해 릿지 회귀(Ridge Regression)의 정규방정식 $W = (X^T X + \alpha I)^{-1} X^T Y$을 즉각 풀어냅니다. 엣지 디바이스(초저전력 임플란트 칩)에 이식하기 가장 좋은 초고속 가볍고 탄탄한 선형 매핑 모델입니다.

### `def fit_pca_gpu(X_tr, n_components=PCA_COMPONENTS)`

- **역할**: 수백 채널에 달하는 시간 축 래깅 피처의 무거운 차원을 다층 퍼셉트론(`MLP`)이 효율적으로 학습할 수 있도록 고속 GPU 기반 저랭크 주성분 분석(`torch.pca_lowrank`)을 통해 핵심 공간 벡터 공간(128차원)으로 차원을 축소합니다.

### `class PeakAwareMLP(nn.Module)`

- **구조적 차별점**: 신경망의 미시적/공간적 정보를 담은 주성분 변환 스트림(`pca_net`)과, 가속도 온셋 지표를 담은 거시 스트림(`peak_net`)을 따로 설계하여 연산한 뒤, 말단에서 결합(`torch.cat`)하는 **듀얼 스트림(Dual-Stream) 신경망** 구조입니다. 고속 주행 시 딥러닝이 방향성을 잃지 않도록 보조합니다.

### `class PeakAwareLoss(nn.Module)` / `class PeakAwarePearsonLoss(nn.Module)`

- **공학적 차별점**: 훈련 중 단순 MSE 손실 함수를 쓰면 모든 오차를 균등하게 취급하므로 데이터 비중이 적은 고속 피크(Peak) 구간의 오차를 모델이 무시하고 학습을 대충 끝내 버립니다.
- **동작 원리**:
    1. 기본 베이스로 아웃라이어 노이즈에 강건한 `HuberLoss`를 장착합니다.
    2. 타겟 속도의 절댓값이 임계치(`8.0 cm/s`)를 넘어서는 급가속 구간일 경우 오차에 배율(`alpha=3.0`)을 적용해 페널티를 강화합니다.
    3. 타겟 속도의 순간 변화량(`torch.diff`)인 가속도 기울기 가중치(`grad_w`)를 추가 곱 연산하여, 방향 변화가 심한 커브나 급출발 시 오차 가중치를 극대화합니다. (`PearsonLoss` 버전은 예측값과 타겟값의 선형 상관성 마저 손실에 직접 가산합니다.)

### `def train_mlp_gpu(X_pca, X_peak, Y_tr, epochs=MLP_EPOCHS)`

- **역할**: 위의 `PeakAwareMLP` 모델을 학습시키는 트레이너 루틴입니다.
- **가속 기법 (`AMP: Automatic Mixed Precision`)**: 카이스트 연구실 급 GPU(A100 등)의 연산 유닛인 텐서 코어(Tensor Core)를 완벽히 깨우기 위해 `torch.cuda.amp.autocast()`와 `GradScaler`를 적용했습니다. 연산 정밀도를 FP32와 FP16으로 자동 스위칭하여 정확도는 유지하되 학습 속도를 2배 이상 끌어올리고 그래픽 메모리를 절약합니다.

### `class LSTMDecoder(nn.Module)` 및 `def train_lstm_gpu(...)`

- **역할**: 시계열의 장기/단기 의존성(Temporal Dynamics)을 포착하는 순단 신경망 모델입니다. 과거 20단계(200ms)의 타임 라인을 순차적으로 훑어 가며 최종 스텝(`out[:, -1, :]`)의 히든 스테이트를 선형 레이어(`fc`)에 통과시켜 속도를 예측합니다.

### `class KalmanDecoder` (메서드: `fit`, `predict`)

- **수학적 배경**: 통제 가능한 상태 궤적과 노이즈 관측 레이어를 분리하는 정통 베이지안 확률 기반 상태공간 모델(State-Space Model)입니다.
- **`fit` 메서드**: 최소자승법(`np.linalg.lstsq`)을 통해 손이 이동하는 물리 법칙을 담은 상태 전이 행렬 $A$와 신경 발화 확률 매핑 행렬 $H$, 그리고 시스템 오차 공분산($Q, R$)을 수학적으로 피팅합니다. 이때 안정적인 수렴을 위해 스파이크 분산 하한선(`spike_var_floor`)을 걸어 수치적 안정성을 꾀합니다.
- **`predict` 메서드**: 정상 상태(Steady-State) 가정을 바탕으로 대수 리카티 방정식(Algebraic Riccati Equation)을 최적화 루프로 사전에 풀어 정적 칼만 이득 변수 `K_steady`를 획득합니다. 이후 루프를 돌며 예측(Prior)-보정(Posterior) 단계를 거쳐 실시간으로 궤적을 트래킹합니다.

## 3. 실행 파이프라인 (Cross-Validation)

### `def run_gpu_decoding(...)`

- **역할**: 4가지 모델의 공정한 비교를 지휘하는 **종합 교차 검증 통제 센터**입니다.
- **핵심 로직 (Time-series Cross Validation)**: 시계열의 연속적 흐름을 보존하기 위해 셔플링을 철저히 차단한 채 `KFold(shuffle=False)`를 실행합니다.
- **예측 궤적의 최종 스무딩**: 모델들이 예측해 낸 거친 날것의 속도 시그널을 생물학적 실제 팔 움직임처럼 매끄럽게 복원하기 위해 최종단에 가우시안 1차원 필터(`gaussian_filter1d`, `sigma=3`) 후처리를 적용해 노이즈를 제거한 뒤 최종 지표를 수집합니다.
